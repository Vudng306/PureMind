const cellText = (nodes: Inline[]): string =>
  nodes.map((n) => ("v" in n ? n.v : "c" in n ? cellText(n.c) : "\n")).join("");
 
/** A short one-line cell ("Case 1", "2.886") is kept on one line rather than wrapped in a narrow column. */
function isLabel(cell: Inline[]): boolean {
  const text = cellText(cell).trim();
  return text.length <= 16 && !text.includes("\n");
}
 
function Inlines({ nodes }: { nodes: Inline[] }) {
  return (
    <>
      {nodes.map((n, i) => {
        switch (n.t) {
          case "text":
            return <Fragment key={i}>{n.v}</Fragment>;
          case "strong":
            return (
              <strong key={i} className="font-semibold text-ink">
                <Inlines nodes={n.c} />
              </strong>
            );
          case "em":
            return (
              <em key={i}>
                <Inlines nodes={n.c} />
              </em>
            );
          case "code":
            return (
              <code key={i} className="rounded bg-soft px-1 py-0.5 font-mono text-[0.85em]">
                {n.v}
              </code>
            );
          case "br":

          case "link":
            return (

                <Inlines nodes={n.c} />
              </a>
            );
        }
      })}
    </>
  );
}

/**
 * An image extracted from the document. It needs the reader's token, which an <img> cannot send, so it is
 * fetched when it scrolls near and shown from a blob URL. A failed one is left out, like a broken web image.
 */
function DocImage({
  documentId,
  name,
  alt,
  className,
}: {
  documentId: string;
  name: string;
  alt: string;
  className: string;
}) {
  const ref = useRef<HTMLDivElement>(null);
  const [src, setSrc] = useState<string | null>(null);
  const [failed, setFailed] = useState(false);
 
  useEffect(() => {
    const el = ref.current;
    if (!el || !documentId) return;
    let url: string | null = null;
    let cancelled = false;
    const observer = new IntersectionObserver(
      (entries) => {
        if (!entries.some((e) => e.isIntersecting)) return;
        observer.disconnect();
        apiFetch(`/documents/${documentId}/images/${name}`)
          .then((res) => res.blob())
          .then((blob) => {
            if (cancelled) return;
            url = URL.createObjectURL(blob);
            setSrc(url);
          })
          .catch(() => !cancelled && setFailed(true));
      },
      { rootMargin: "800px 0px" },
    );
    observer.observe(el);
    return () => {
      cancelled = true;
      observer.disconnect();
      if (url) URL.revokeObjectURL(url);
    };
  }, [documentId, name]);
 
  if (failed) return null;
  return src ? (
    // eslint-disable-next-line @next/next/no-img-element
    <img src={src} alt={alt} className={className} />
  ) : (
    <div ref={ref} className={`${className} min-h-24 min-w-24 animate-pulse bg-soft`} aria-hidden />
  );
}
 
/** A figure of the document: extracted from the file (`pm-image:`) or linked from the web article. */
export function FigureImage({
  documentId,
  src,
  alt,
  className = "mx-auto max-h-[70vh] w-auto max-w-full rounded-lg",
}: {
  documentId: string;
  src: string;
  alt: string;
  className?: string;
}) {
  const own = DOC_IMAGE.exec(src);
  if (own) return <DocImage documentId={documentId} name={own[1]} alt={alt} className={className} />;
  return (
    // Articles link images on any host; next/image would need every one of them in an allowlist.
    // eslint-disable-next-line @next/next/no-img-element
    <img
      src={src}
      alt={alt}
      loading="lazy"
      // The source site learns nothing about who is reading, and referer-based hotlink blocks do not trigger.
      referrerPolicy="no-referrer"
      className={className}
      onError={(e) => {
        // A removed or blocked image would otherwise leave a broken icon in the middle of the text.
        e.currentTarget.closest("figure")?.setAttribute("hidden", "");
      }}
    />
  );
}

/** "View in the original" on a figure or table: the extraction may have misread it, the PDF has it right. */
function SourceButton({ blockId }: { blockId: string }) {

function Figure({ blockId, src, alt }: { blockId: string; src: string; alt: string }) {
  const t = useT();
  const { documentId, onExplain, charts } = useContext(ImagesContext);
  const chart = charts?.[DOC_IMAGE.exec(src)?.[1] ?? ""];

        <div className="absolute right-2 top-2 flex flex-wrap justify-end gap-1.5">
          <SourceButton blockId={blockId} />
          {chart && (
            <button
              type="button"
              aria-expanded={showData}
              onClick={() => setShowData((v) => !v)}

              <Icon name="chart" size={13} className="text-accent-text" />
              {t("reader.chartData")}

            <button type="button" onClick={() => onExplain(src, alt)} className={HOVER_CHIP}>

              {t("chat.explainImage")}

  );
}
 
const BlockView = memo(function BlockView({ block }: { block: Block }) {

  switch (block.t) {

      );

      const ListTag = block.ordered ? "ol" : "ul";

          {block.items.map((item, i) => (
            <li key={i} style={{ marginLeft: `${item.depth * 1.25}em` }}>
              <Inlines nodes={item.c} />
            </li>
          ))}
        </ListTag>
      );
    }
    case "table":
      return (
        <div {...common} className="group relative mb-[18px]">
          <div className="absolute right-1 top-1 z-[1] flex">
            <SourceButton blockId={block.id} />
          </div>

            <table className="w-full border-collapse font-sans text-[0.8em]">
              <tbody>
                {block.rows.map((row, r) => (

                      <td
                        key={c}

                      >

                    ))}

      );
    case "image":
      return (
        <figure {...common} className="mb-[18px] select-none">
          <Figure blockId={block.id} src={block.src} alt={block.alt} />
          {block.alt && (
            <figcaption className="mt-1.5 text-center font-sans text-[0.72em] leading-normal text-muted">
              {block.alt}
            </figcaption>
          )}

      );
    case "math":
      return (
        // Not selectable: the typeset glyphs are not the equation's text, so a highlight could not hold.

      );
  }
});
 
/** Trim whitespace at both ends of a [start, end) slice of `text`. */
function trimSpan(text: string, start: number, end: number): [number, number] {
  while (start < end && /\s/.test(text[start])) start++;
  while (end > start && /\s/.test(text[end - 1])) end--;
  return [start, end];
}

  flashHighlightId?: string | null;
  onSelect: (selection: TextSelection | null) => void;
  onHighlightClick: (id: string) => void;
  /** Highlights not placed in the current text, and why: several places fit, or none does (AN-104). */
  onUnanchored?: (issues: Record<string, AnchorIssue>) => void;
  onScrollProgress: (state: ScrollState) => void;
  /** Ask the AI to explain a figure (the button on each image). */
  onExplainImage?: (src: string, alt: string) => void;
  /** Show a block on the original PDF (the button on figures and tables). */
  onViewSource?: (blockId: string) => void;
  hasSource?: (blockId: string) => boolean;
  /** The data of the vector charts (the "Chart data" button on those figures). */
  charts?: Record<string, Chart>;
  bottomPadding: number;
  /** How much of the top of the text a bar lying over it covers. */
  topInset?: number;
  /** A phone: the selection follows the system's handles and stays while the text is scrolled. */
  compact?: boolean;
}) {
  const t = useT();
  const lang = useLang();
  const images = useMemo(
    () => ({ documentId, onExplain: onExplainImage, onViewSource, hasSource, charts }),
    [documentId, onExplainImage, onViewSource, hasSource, charts],
  );
  const scrollerRef = useRef<HTMLDivElement>(null);
  const articleRef = useRef<HTMLElement>(null);
  const { query, index, onResult } = search;
 
  // ---- find
  useEffect(() => {
    const article = articleRef.current;
    if (!article || !query.trim()) {
      clearHighlights();
      onResult(0, query);
      return;
    }
    const ranges = findRanges(article, query);
    onResult(ranges.length, query);
    const current = ranges.length ? ranges[((index % ranges.length) + ranges.length) % ranges.length] : null;
    paintHighlights(ranges, current);
    if (current && scrollerRef.current) scrollRangeIntoView(scrollerRef.current, current);
  }, [query, index, blocks, onResult]);
 
  useEffect(() => clearHighlights, []);
 
  // ---- highlights (placed where their passage is now, and only where that is sure; see locate)
  const anchors = useRef(new Map<string, Anchor[]>());
  useEffect(() => {
    const article = articleRef.current;
    if (!article) return;
    let active: Range[] = [];
    let flash: Range[] = [];
    const painted: { color: Highlight["color"]; range: Range }[] = [];
    const issues: Record<string, AnchorIssue> = {};
    let texts: TextBlock[] | null = null;
    const blocksNow = () => (texts ??= readBlocks(article));
    anchors.current = new Map();
    for (const h of highlights) {
      const placed = placeHighlight(article, h, blocksNow);
      const spots = placed.spot ? (placed.spots ?? [placed.spot]) : [];
      const ranges = spots.map((s) => rangeAt(article, s)).filter((r) => r !== null);
      if (!ranges.length) {
        // A highlight that is not sure of its place is painted nowhere (TSD §5.2). One still on its way to
        // the server has not been told its place yet.
        if (h.save !== "saving") issues[h.id] = placed.state === "review" ? "review" : "lost";
        continue;
      }
      anchors.current.set(h.id, spots);
      for (const range of ranges) painted.push({ color: h.color, range });
      if (h.id === activeHighlightId) active = ranges;
      if (h.id === flashHighlightId) flash = ranges;
    }
    paintAnnotations(painted, active, flash);
    onUnanchored?.(issues);
  }, [highlights, activeHighlightId, flashHighlightId, blocks, onUnanchored]);
 
  useEffect(() => clearAnnotations, []);
 
  // ---- the place by block (RD-107): kept across a change of font, width or window, which a fraction is not
  const blockEls = useRef<HTMLElement[] | null>(null);
  useEffect(() => {
    blockEls.current = null;
  }, [blocks]);
  const readPlace = useCallback((): ReadingPlace | null => {
    const scroller = scrollerRef.current;
    const article = articleRef.current;
    if (!scroller || !article) return null;
    const els = (blockEls.current ??= [...article.querySelectorAll<HTMLElement>(":scope > [data-block-id]")]);
    if (!els.length) return null;
    const line = scroller.getBoundingClientRect().top + LINE + topInset;
    // The last block that starts at or above the line (blocks are in order down the page).
    let lo = 0;
    let hi = els.length - 1;
    while (lo < hi) {
      const mid = (lo + hi + 1) >> 1;
      if (els[mid].getBoundingClientRect().top <= line) lo = mid;
      else hi = mid - 1;
    }
    const rect = els[lo].getBoundingClientRect();
    const offset = rect.height > 0 ? Math.max(0, (line - rect.top) / rect.height) : 0;
    return { blockId: els[lo].dataset.blockId ?? "", offset };
  }, [topInset]);
  const goToPlace = useCallback(
    ({ blockId, offset }: ReadingPlace) => {
      const el = articleRef.current?.querySelector<HTMLElement>(`:scope > [data-block-id="${CSS.escape(blockId)}"]`);
      const scroller = scrollerRef.current;
      if (!el || !scroller) return false;
      const rect = el.getBoundingClientRect();
      const top = rect.top - scroller.getBoundingClientRect().top + scroller.scrollTop - LINE - topInset + offset * rect.height;
      // A place that is already under the line is left alone, so nothing moves by a rounding.
      if (Math.abs(top - scroller.scrollTop) >= 1) scroller.scrollTop = top;
      return true;
    },
    [topInset],
  );
 
  // A jump that glides to its passage: where it is heading, until it gets there or the reader takes the scroll.
  const heading = useRef<{ top: number; since: number } | null>(null);
  const glideTo = useCallback((scroller: HTMLElement, top: number) => {
    const max = scroller.scrollHeight - scroller.clientHeight;
    heading.current = { top: Math.min(Math.max(0, top), max), since: performance.now() };
    scroller.scrollTo({ top, behavior: "smooth" });
  }, []);
  const takeOver = useCallback(() => {
    heading.current = null;
  }, []);
  useEffect(() => {
    const scroller = scrollerRef.current;
    scroller?.addEventListener("scrollend", takeOver);
    return () => scroller?.removeEventListener("scrollend", takeOver);
  }, [takeOver]);
 
  useImperativeHandle(
    handleRef,
    () => ({
      scrollToBlock(blockId) {
        const el = articleRef.current?.querySelector<HTMLElement>(`[data-block-id="${CSS.escape(blockId)}"]`);
        const scroller = scrollerRef.current;
        if (!el || !scroller) return;
        const top = el.getBoundingClientRect().top - scroller.getBoundingClientRect().top + scroller.scrollTop - 24 - topInset;
        glideTo(scroller, top);
      },
      scrollToHighlight(h) {
        const article = articleRef.current;
        const scroller = scrollerRef.current;
        const spot = article && placeHighlight(article, h).spot;
        const range = article && spot && rangeAt(article, spot);
        if (!range || !scroller) return false;
        const rect = range.getBoundingClientRect();
        const box = scroller.getBoundingClientRect();
        glideTo(scroller, scroller.scrollTop + rect.top - box.top - box.height / 3);
        return true;
      },
      jumping() {
        const scroller = scrollerRef.current;
        const jump = heading.current;
        if (!scroller || !jump) return false;
        // "scrollend" says when the glide is over; without it (older Safari) a glide is given a few seconds.
        const patience = "onscrollend" in window ? 20000 : 4000;
        if (Math.abs(scroller.scrollTop - jump.top) <= 2 || performance.now() - jump.since > patience) heading.current = null;
        return heading.current !== null;
      },
      scrollToFraction(fraction) {
        const scroller = scrollerRef.current;
        if (scroller) scroller.scrollTop = fraction * (scroller.scrollHeight - scroller.clientHeight);
      },
      scrollToPlace: goToPlace,
      place: () => readPlaceRef.current(),
      currentSelection: () => readSelectionRef.current(),
    }),
    [glideTo, goToPlace, topInset],
  );
 
  // ---- selection → highlight menu; click on a highlight → open it in the panel
  const readSelection = useCallback(() => {
    const article = articleRef.current;
    const sel = window.getSelection();
    if (!article || !sel || sel.rangeCount === 0 || sel.isCollapsed) return null;
    const range = sel.getRangeAt(0);
    if (!article.contains(range.commonAncestorContainer)) return null;
    const block = blockOf(range.startContainer);
    if (!block?.dataset.blockId) return null;
    const blockText = block.textContent ?? "";
    let start = offsetIn(block, range.startContainer, range.startOffset);
    // A selection running into later blocks is clamped to the first one.
    let end = block.contains(range.endContainer) ? offsetIn(block, range.endContainer, range.endOffset) : blockText.length;
    [start, end] = trimSpan(blockText, start, end);
    if (end <= start) return null;
    const text = blockText.slice(start, end);
    const exact = rangeFor(article, { blockId: block.dataset.blockId, start, end, text });
    return {
      blockId: block.dataset.blockId,
      start,
      end,
      canonStart: canonLength(blockText.slice(0, start)),
      canonEnd: canonLength(blockText.slice(0, start)) + canonLength(text),
      text,
      ...contextOf(blockText, start, end),
      rect: (exact ?? range).getBoundingClientRect(),
    } satisfies TextSelection;
  }, []);
 
  useSelectionTracking(compact, readSelection, onSelect);
 
  const readPlaceRef = useRef(readPlace);
  readPlaceRef.current = readPlace;
  const readSelectionRef = useRef(readSelection);
  readSelectionRef.current = readSelection;
 
  const onPointerUp = useCallback(
    (e: React.MouseEvent | React.TouchEvent) => {
      // Let the browser finish updating the selection first.
      requestAnimationFrame(() => {
        const selection = readSelection();
        if (selection) return onSelect(selection);
        onSelect(null);
        if (!("clientX" in e)) return;
        const caret = caretAt(e.clientX, e.clientY);
        const block = caret && blockOf(caret.node);
        if (!caret || !block?.dataset.blockId) return;
        const offset = offsetIn(block, caret.node, caret.offset);
        const blockId = block.dataset.blockId;
        // Of highlights that overlap, the shortest: the longer one can be clicked outside it.
        const size = (spots: Anchor[]) => spots.reduce((sum, a) => sum + a.end - a.start, 0);
        const [hit] = [...anchors.current]
          .filter(([, spots]) => spots.some((a) => a.blockId === blockId && a.start <= offset && offset < a.end))
          .sort((a, b) => size(a[1]) - size(b[1]));
        if (hit) onHighlightClick(hit[0]);
      });
    },
    [readSelection, onSelect, onHighlightClick],
  );
 
  // Keyboard selection (Shift+arrows) also opens the menu.
  const onKeyUp = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.shiftKey || e.key === "Shift") onSelect(readSelection());
    },
    [readSelection, onSelect],
  );
 
  // ---- progress & current page
  const frame = useRef(0);
  // Turning the phone or resizing the window rewraps the text under an unchanged scrollTop. What was being read
  // goes back under the line, before the scroll this causes can be taken, and saved, as the reader's own.
  const held = useRef<{ width: number; place: ReadingPlace | null }>({ width: 0, place: null });
  const refit = useCallback(() => {
    const width = scrollerRef.current?.clientWidth ?? 0;
    if (!width || width === held.current.width) return false;
    const { width: before, place } = held.current;
    held.current.width = width;
    if (!before || !place || heading.current) return false;
    goToPlace(place);
    return true;
  }, [goToPlace]);
  useEffect(() => {
    const scroller = scrollerRef.current;
    if (!scroller) return;
    const watch = new ResizeObserver(() => refit());
    watch.observe(scroller);
    return () => watch.disconnect();
  }, [refit]);
 
  const onScroll = useCallback(() => {
    cancelAnimationFrame(frame.current);
    frame.current = requestAnimationFrame(() => {
      const scroller = scrollerRef.current;
      const article = articleRef.current;
      if (!scroller || !article || refit()) return;
      held.current.place = readPlace();
      const max = scroller.scrollHeight - scroller.clientHeight;
      const fraction = max > 0 ? Math.min(1, Math.max(0, scroller.scrollTop / max)) : 1;
      let page: number | null = null;
      const breaks = article.querySelectorAll<HTMLElement>("[data-page-break]");
      if (breaks.length) {
        page = 1;
        const line = scroller.getBoundingClientRect().top + scroller.clientHeight * 0.35;
        for (const b of breaks) {
          if (b.getBoundingClientRect().top > line) break;
          page = Number(b.dataset.pageBreak);
        }
      }
      onScrollProgress({ fraction, page, place: held.current.place, top: scroller.scrollTop, view: scroller.clientHeight });
      // On a phone the selection bar is not over the text, and the passage may be scrolled clear of it.
      if (!compact) onSelect(null);
    });
  }, [onScrollProgress, onSelect, readPlace, refit, compact]);
 
  useEffect(() => () => cancelAnimationFrame(frame.current), []);
 
  return (
    <div
      ref={scrollerRef}
      onScroll={onScroll}
      onWheel={takeOver}
      onTouchStart={takeOver}
      onPointerDown={takeOver}
      onKeyDown={takeOver}
      className="h-full overflow-y-auto"
    >
      <article
        ref={articleRef}
        onMouseUp={onPointerUp}
        onTouchEnd={onPointerUp}
        onKeyUp={onKeyUp}
        className={`mx-auto break-words px-6 pt-10 text-body sm:pt-14 ${fontFamily === "sans" ? "font-sans" : "font-serif"}`}
        style={{ fontSize, lineHeight, maxWidth: width + 48, paddingBottom: bottomPadding, paddingTop: topInset ? 24 + topInset : undefined }}
        lang={lang}
      >
        {header}
        <ImagesContext.Provider value={images}>
          {blocks.map((b) => (
            <BlockView key={b.id} block={b} />
          ))}
        </ImagesContext.Provider>
        {footer}
        <p className="mt-11 select-none font-sans text-[13px] text-muted">{t("reader.tip")}</p>
      </article>
    </div>
  );
}
