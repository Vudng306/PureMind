/**

function PrintSheet({
  title,
  content,
  sources,
  groups,
}: {
  title: string;
  content: string;
  sources: NotebookSource[];
  groups: SourceGroup[];
}) {
  const t = useT();
  return (
    <div className="print-sheet hidden print:block">
      <h1 className="font-serif text-[22pt] font-medium leading-tight text-ink">{title}</h1>
      <p className="mb-6 mt-2 border-b border-line pb-3 text-[9pt] text-muted">
        {t("nb.printedFrom", { date: formatDate(new Date().toISOString()) })}
        {sources.length > 0 && ` · ${t("nb.meta", { n: sources.length, docs: groups.length })}`}
      </p>
      <NotebookMarkdown
        content={content}
        sources={sources}
        title={title}
        className="font-serif text-[11.5pt] leading-[1.6] text-body"
      />
      {groups.length > 0 && (
        <section className="break-before-page">
          <h2 className="mb-5 font-serif text-[16pt] font-semibold text-ink">{t("nb.modeDigest")}</h2>
          <HighlightDigest groups={groups} />
        </section>
      )}
    </div>
  );

/** The download button: the notebook as a Word file (to read and print) or as Markdown (for other note apps).
 * Closes on Escape, on a click outside and after a choice. */
function ExportMenu({ disabled, onExport }: { disabled: boolean; onExport: (format: ExportFormat) => void }) {
  const t = useT();
  const [open, setOpen] = useState(false);
  const root = useRef<HTMLDivElement>(null);
  const button = useRef<HTMLButtonElement>(null);

  useEffect(() => {
    if (!open) return;
    root.current?.querySelector<HTMLElement>("[role=menuitem]")?.focus();
    const onDown = (e: PointerEvent) => {
      if (!root.current?.contains(e.target as Node)) setOpen(false);
    };
    document.addEventListener("pointerdown", onDown);
    return () => document.removeEventListener("pointerdown", onDown);
  }, [open]);

  function onKeyDown(e: React.KeyboardEvent) {
    if (e.key === "Escape" && open) {
      setOpen(false);
      button.current?.focus();
    } else if (open && (e.key === "ArrowDown" || e.key === "ArrowUp")) {
      e.preventDefault();
      const items = [...(root.current?.querySelectorAll<HTMLElement>("[role=menuitem]") ?? [])];
      const i = items.indexOf(document.activeElement as HTMLElement);
      items[(i + (e.key === "ArrowDown" ? 1 : items.length - 1)) % items.length]?.focus();
    }
  }

  const choose = (format: ExportFormat) => () => {
    setOpen(false);
    onExport(format);
  };
  const item = "flex w-full items-start gap-2.5 rounded-lg px-3 py-2 text-left hover:bg-soft focus:bg-soft focus:outline-none";

  return (
    <div ref={root} className="relative max-md:order-3" onKeyDown={onKeyDown}>
      <button
        ref={button}
        type="button"
        className="icon-btn h-10 w-10 max-md:h-11 max-md:w-11"
        aria-label={t("nb.export")}
        title={t("nb.export")}
        aria-haspopup="menu"
        aria-expanded={open}
        disabled={disabled}
        onClick={() => setOpen(!open)}
      >
        <Icon name="download" />
      </button>
      {open && (
        <div
          role="menu"
          aria-label={t("nb.export")}
          className="absolute right-0 top-12 z-30 flex w-64 flex-col gap-0.5 rounded-xl border border-line bg-surface p-1.5 text-ink shadow-float"
        >
          <button type="button" role="menuitem" className={item} onClick={choose("docx")}>
            <Icon name="file" size={16} className="mt-0.5 shrink-0" />
            <span className="flex flex-col">
              <span className="text-sm font-medium">{t("nb.exportDocx")}</span>
              <span className="text-xs text-muted">{t("nb.exportDocxHint")}</span>
            </span>
          </button>
          <button type="button" role="menuitem" className={item} onClick={choose("md")}>
            <Icon name="note" size={16} className="mt-0.5 shrink-0" />
            <span className="flex flex-col">
              <span className="text-sm font-medium">{t("nb.exportMd")}</span>
              <span className="text-xs text-muted">{t("nb.exportMdHint")}</span>
            </span>
          </button>
        </div>
      )}
    </div>
  );
}

function VersionsDialog({

                <Spinner />

                <Alert>{version.error instanceof Error ? version.error.message : MSG["MSG-99"]}</Alert>

}

  const tooLong = content.length > MAX_NOTEBOOK_CHARS;

  const valid = titleOk && !tooLong;

  const send = useCallback(
    async (kind: "auto" | "save") => {
      const snapshot = latest.current;
      const patch: NotebookPatch = { title: snapshot.title, content: snapshot.content, topic: snapshot.topic };
      if (kind === "save") patch.status = "saved";
      setSaving(kind);
      try {
        const nb = await mutateAsync(patch);
        stored.current = snapshot;
        setSavedAt(nb.updated_at);
        setAutoFailed(false);
        if (kind === "save") {
          setSaveError(null);
          showToast(t("nb.saved"));
        }
      } catch (e) {
        if (kind === "save") setSaveError(e instanceof ApiError && e.status === 422 ? e.message : MSG["MSG-99"]);
        else setAutoFailed(true);
      } finally {
        setSaving(null);
      }
    },
    [mutateAsync, showToast, t],
  );

  // Autosave: 5 seconds after the first unsaved change, then again while changes keep coming (the status stays).
  useEffect(() => {
    if (!dirty || !valid || saving) return;
    const t = setTimeout(() => void send("auto"), AUTOSAVE_MS);
    return () => clearTimeout(t);
  }, [dirty, valid, saving, send]);

  // Warn before leaving with unsaved changes.
  useEffect(() => {
    if (!dirty) return;
    const onBeforeUnload = (e: BeforeUnloadEvent) => e.preventDefault();
    window.addEventListener("beforeunload", onBeforeUnload);
    return () => window.removeEventListener("beforeunload", onBeforeUnload);
  }, [dirty]);

  // Leaving inside the app: send what is left, in the background.
  const updateRef = useRef(update);
  updateRef.current = update;
  useEffect(
    () => () => {
      const { title: t, content: c, topic: o } = latest.current;
      const changed = t !== stored.current.title || c !== stored.current.content || o !== stored.current.topic;
      if (changed && !deleted.current && t.trim() && c.length <= MAX_NOTEBOOK_CHARS) {
        updateRef.current.mutate({ title: t, content: c, topic: o });
      }
    },
    [],
  );

  const sources = notebook.sources;
  const sourceIds = useMemo(() => new Set(sources.map((s) => s.highlight.id)), [sources]);
  const groups = useMemo(() => groupSources(sources), [sources]);
  const printContent = useDeferredValue(content); // the hidden print sheet never slows typing down
  const missing = useMemo(() => {
    const have = new Set(sources.map((s) => s.position));
    return citedPositions(content)
      .filter((n) => !have.has(n))
      .sort((a, b) => a - b);
  }, [content, sources]);

  async function setSources(ids: string[], done?: () => void) {
    setSaveError(null);
    try {
      await mutateAsync({ highlight_ids: ids });
      done?.();
    } catch (e) {
      setSaveError(e instanceof ApiError ? e.message : MSG["MSG-99"]);
    }
  }

  const removeSource = (source: NotebookSource) =>
    void setSources(sources.filter((x) => x !== source).map((x) => x.highlight.id));

  /** Puts text in at the caret when the editor is open, at the end otherwise, then shows it in the editor. */
  function insertText(text: string) {
    const area = editorRef.current;
    const at = view === "edit" && area ? area.selectionEnd : content.length;
    const before = content.slice(0, at).replace(/\s*$/, "");
    const after = content.slice(at).replace(/^\s*/, "");
    setContent(`${before}${before ? "\n\n" : ""}${text}${after ? `\n\n${after}` : "\n"}`);
    setView("edit");
  }

  /** The passage as a quote with its [n], which the preview links back to the place in the document. */
  const quoteSource = (source: NotebookSource) =>
    insertText(`> ${source.highlight.selected_text.replace(/\s*\n+\s*/g, " ").trim()} [${source.position}]`);

  /** Later posts from this notebook then say that AI helped. */
  function insertSummary(markdown: string, aiModel: string) {
    insertText(markdown);
    if (!notebook.ai_model) {
      mutateAsync({ ai_model: aiModel }).catch(() => setSaveError(MSG["MSG-99"]));
    }
  }

  /** The notebook as it stands in the editor, with the passages its [n] cite: a Word file laid out like the
   * page (title, the line under it, the notebook's headings), or the Markdown itself. */
  async function exportNotebook(format: ExportFormat) {
    const now = new Date();
    const name = exportFileName(title.trim() || null, now, "puremind-notebook", format);
    try {
      if (format === "docx") {
        const { notebookDocxBlob } = await import("@/lib/notebook-docx");
        const madeOn = t("nb.madeOn", { date: formatDate(notebook.created_at) });
        // The same two lines as under the title on the page, in the same order.
        const meta = [
          topic.trim() ? `${t("nb.topicLabel")}: ${topic.trim()}` : "",
          sources.length > 0 ? `${madeOn} · ${t("nb.meta", { n: sources.length, docs: groups.length })}` : madeOn,
        ];
        const blob = await notebookDocxBlob({ title, meta, content, sources, lang, origin: window.location.origin, now });
        downloadBlob(blob, name);
      } else {
        downloadText(notebookExportMarkdown(title, content, sources, now, lang, window.location.origin), name);
      }
      showToast(t("nb.exported"));
    } catch {
      setSaveError(MSG["MSG-99"]);
    }
  }

  // The browser names a printed PDF after the page title.
  useEffect(() => {
    const before = document.title;
    if (title.trim()) document.title = `${title.trim()} · PureMind`;
    return () => {
      document.title = before;
    };
  }, [title]);

  async function doDelete() {
    try {
      deleted.current = true;
      await remove.mutateAsync(notebook.id);
      showToast(t("nb.deleted"));
      router.replace("/notebooks/archive");
    } catch {
      deleted.current = false;
      setConfirmDelete(false);
      setSaveError(MSG["MSG-99"]);
    }
  }

  const status = saving
    ? t("nb.statusSaving")
    : autoFailed
      ? t("nb.statusFailed")
      : dirty
        ? t("nb.statusDirty")
        : t("nb.statusSavedAt", { time: timeOf(savedAt, lang) });

  return (
    <>
      <div className="mx-auto flex max-w-[1180px] flex-col gap-5 print:hidden">
        {/* On a phone: the way back and the actions on one line, what state the notebook is in on the next. */}
        <div className="flex flex-wrap items-center gap-2.5 max-md:gap-x-1 max-md:gap-y-0.5">
          <Link href="/notebooks/archive" className="flex items-center gap-1 text-sm text-muted hover:text-ink max-md:order-1 max-md:h-11 max-md:pr-2">
            <Icon name="back" size={16} />
            <span className="max-[359px]:sr-only">{t("notebooks.title")}</span>
          </Link>
          <span
            className={`flex h-6 items-center rounded-full px-2 text-xs font-semibold max-md:order-5 max-md:mr-1.5 ${
              notebook.status === "saved" ? "bg-success-soft text-success" : "bg-soft text-ink"
            }`}
          >
            {t(STATUS_LABEL[notebook.status])}
          </span>
          <span className={`text-xs max-md:order-5 ${autoFailed ? "text-danger" : "text-muted"}`} role="status">
            {status}
          </span>
          <span className="flex-1 max-md:order-2" />
          <span aria-hidden className="hidden basis-full max-md:order-4 max-md:block" />
          <button
            type="button"
            className="icon-btn h-10 w-10 max-md:order-3 max-md:h-11 max-md:w-11"
            aria-label={t("nb.print")}
            title={t("nb.print")}
            onClick={() => window.print()}
          >
            <Icon name="print" />
          </button>
          <ExportMenu disabled={!content.trim()} onExport={(f) => void exportNotebook(f)} />
          <button
            type="button"
            className="icon-btn h-10 w-10 max-md:order-3 max-md:h-11 max-md:w-11"
            aria-label={t("notebooks.versions")}
            title={t("notebooks.versions")}
            onClick={() => setHistory(true)}
          >
            <Icon name="history" />
          </button>
          <button
            type="button"
            className="icon-btn h-10 w-10 max-md:order-3 max-md:h-11 max-md:w-11"
            aria-label={t("nb.deleteNotebook")}
            title={t("nb.deleteNotebook")}
            onClick={() => setConfirmDelete(true)}
          >
            <Icon name="trash" />
          </button>
          <button type="button" className="btn-primary h-10 max-md:order-3 max-md:ml-1 max-md:h-11" onClick={() => void send("save")} disabled={!valid || saving !== null}>
            {t(saving === "save" ? "common.saving" : "common.save")}
          </button>
        </div>

        {saveError && <Alert>{saveError}</Alert>}

        <div className={view === "digest" ? "mx-auto w-full max-w-[880px]" : "grid grid-cols-[minmax(0,1fr)] gap-8 lg:grid-cols-[minmax(0,1fr)_340px]"}>
          <div className="flex min-w-0 flex-col gap-3">
            <div className="rounded-2xl border border-line bg-surface">
              <div className="flex flex-col gap-2 px-4 pt-5 sm:px-9 sm:pt-8">
                <label className="flex flex-col gap-1">
                  <span className="sr-only">{t("nb.titleLabel")}</span>
                  <input
                    value={title}
                    onChange={(e) => setTitle(e.target.value)}
                    maxLength={MAX_NOTEBOOK_TITLE}
                    className="field-large w-full rounded-lg border border-transparent bg-transparent px-1 py-1 font-serif text-3xl font-medium leading-tight text-ink outline-none hover:border-line focus:border-focus sm:text-4xl"
                  />
                  {!titleOk && <span className="px-1 text-sm text-danger">{t("nb.titleRequired")}</span>}
                </label>
                <label className="flex items-center gap-2 px-1 text-[13px] text-muted">
                  <span className="shrink-0">{t("nb.topicLabel")}</span>
                  <input
                    value={topic}
                    maxLength={MAX_NOTEBOOK_TOPIC}
                    placeholder={t("nb.topicPlaceholder")}
                    onChange={(e) => setTopic(e.target.value)}
                    className="h-8 min-w-0 flex-1 rounded-lg border border-transparent bg-transparent px-1.5 text-[13px] text-ink outline-none hover:border-line focus:border-focus max-md:h-11 sm:max-w-[280px]"
                  />
                </label>
                <span className="px-1 text-[13px] text-muted">
                  {t("nb.madeOn", { date: formatDate(notebook.created_at) })}
                  {sources.length > 0 && ` · ${t("nb.meta", { n: sources.length, docs: groups.length })}`}
                </span>
              </div>

              <div className="mt-4 flex gap-1 overflow-x-auto border-b border-line px-3 [scrollbar-width:none] sm:px-8" role="group" aria-label={t("nb.mode")}>
                {VIEWS.map(([value, label]) => (
                  <button
                    key={value}
                    type="button"
                    aria-pressed={view === value}
                    onClick={() => setView(value)}
                    className={`flex h-11 shrink-0 items-center gap-1.5 border-b-2 px-3 text-sm ${
                      view === value ? "border-accent font-semibold text-ink" : "border-transparent text-muted hover:text-ink"
                    }`}
                  >
                    {t(label)}
                    {value === "digest" && sources.length > 0 && (
                      <span className="rounded-full bg-soft px-1.5 text-xs font-semibold text-ink">{sources.length}</span>
                    )}
                  </button>
                ))}
              </div>

              <div className="px-5 pb-7 pt-6 sm:px-10 sm:pb-10 sm:pt-8">
                {view === "edit" ? (
                  <div className="flex flex-col gap-1.5">
                    <label htmlFor="notebook-content" className="sr-only">
                      {t("nb.markdownContent")}
                    </label>
                    <textarea
                      id="notebook-content"
                      ref={editorRef}
                      value={content}
                      onChange={(e) => setContent(e.target.value)}
                      spellCheck={false}
                      className="min-h-[60vh] w-full resize-y rounded-xl border border-field bg-bg/40 p-4 font-mono text-[13.5px] leading-relaxed text-ink outline-none focus:border-focus max-md:min-h-[40dvh] max-md:p-3"
                    />
                    <span className={`flex flex-wrap gap-x-3 text-xs ${tooLong ? "text-danger" : "text-muted"}`}>
                      <span>
                        {t("nb.charCount", {
                          n: content.length.toLocaleString(lang),
                          max: MAX_NOTEBOOK_CHARS.toLocaleString(lang),
                        })}
                        {tooLong ? ` — ${MSG["MSG-29"]}` : ""}
                      </span>
                      <span>{t("nb.markdownHint")}</span>
                    </span>
                  </div>
                ) : view === "digest" ? (
                  <div className="flex flex-col gap-7">
                    <div className="flex flex-wrap items-center gap-2.5">
                      <p className="min-w-0 flex-1 basis-[260px] text-sm text-muted">
                        {sources.length > 0 ? t("nb.digestLede", { n: sources.length, docs: groups.length }) : t("nb.noSources")}
                      </p>
                      <button
                        type="button"
                        className="btn-outline h-10 px-3.5 text-sm max-md:h-11"
                        onClick={() => setAdding(true)}
                        disabled={update.isPending}
                      >
                        <Icon name="plus" size={16} />
                        {t("nb.addSources")}
                      </button>
                    </div>
                    <HighlightDigest groups={groups} onRemove={removeSource} busy={update.isPending} />
                  </div>
                ) : content.trim() ? (
                  <NotebookMarkdown content={content} sources={sources} title={title} />
                ) : (
                  <p className="text-muted">{t("nb.emptyNotebook")}</p>
                )}
              </div>
            </div>
            <span className="px-1 text-xs text-muted">
              {t("nb.aiCaveat")}
            </span>
          </div>

          {view !== "digest" && (
            <aside
              className="flex flex-col gap-3 lg:sticky lg:top-[96px] lg:max-h-[calc(100vh-120px)] lg:self-start lg:overflow-y-auto"
              aria-label={t("nb.sourcesAria")}
            >
              <NotebookDocuments notebook={notebook} onInsert={insertSummary} />
              <div className="mt-2 flex items-center gap-2">
                <h2 className="eyebrow flex-1">{t("nb.sourcesCount", { n: sources.length })}</h2>
                <button type="button" className="btn-ghost h-9 px-3 text-sm max-md:h-11" onClick={() => setAdding(true)} disabled={update.isPending}>
                  <Icon name="plus" size={16} />
                  {t("nb.addSources")}
                </button>
              </div>
              {sources.length === 0 && missing.length === 0 && <p className="text-sm text-muted">{t("nb.noSources")}</p>}
              <SourceGroups groups={groups} onRemove={removeSource} onInsert={quoteSource} busy={update.isPending} />
              {missing.length > 0 && (
                <ul className="flex flex-col gap-2">
                  {missing.map((n) => (
                    <li key={n} className="flex items-center gap-2.5 rounded-xl border border-dashed border-line p-3 text-sm text-muted">
                      <span className="flex h-6 min-w-6 items-center justify-center rounded bg-soft px-1 text-xs font-semibold line-through">
                        {n}
                      </span>
                      {t("nb.removedSource")}
                    </li>
                  ))}
                </ul>
              )}
            </aside>
          )}
        </div>

        <AddSourcesDialog
          open={adding}
          existing={sourceIds}
          busy={update.isPending}
          onClose={() => setAdding(false)}
          onAdd={(ids) => void setSources([...sources.map((s) => s.highlight.id), ...ids], () => setAdding(false))}
        />

        <VersionsDialog
          open={history}
          notebookId={notebook.id}
          sources={sources}
          onClose={() => setHistory(false)}
          onRestore={(v) => {
            setTitle(v.title);
            setContent(v.content);
            setView("preview");
            setHistory(false);
            showToast(t("nb.restored"));
          }}
        />

        <ConfirmDialog
          open={confirmDelete}
          title={t("notebooks.deleteTitle")}
          busy={remove.isPending}
          onConfirm={() => void doDelete()}
          onCancel={() => setConfirmDelete(false)}
        >
          {t("nb.deleteBody", { title: notebook.title })}
        </ConfirmDialog>
      </div>

      <PrintSheet title={title} content={printContent} sources={sources} groups={groups} />
    </>
  );
}

export default function NotebookPage() {
  const t = useT();
  const { id } = useParams<{ id: string }>();
  const query = useNotebook(id);

  if (query.isPending) return <Spinner />;
  if (query.isError) {
    const notFound = query.error instanceof ApiError && query.error.status === 404;
    return (
      <div className="mx-auto flex max-w-[880px] flex-col items-start gap-3">
        <Alert>
          {notFound ? t("nb.notFound") : query.error instanceof Error ? query.error.message : MSG["MSG-99"]}
        </Alert>
        {notFound ? (
          <Link href="/notebooks/archive" className="btn-outline">
            {t("nb.backToList")}
          </Link>
        ) : (
          <button type="button" className="btn-outline" onClick={() => query.refetch()}>
            {t("common.retry")}
          </button>
        )}
      </div>
    );
  }
  return <Editor key={query.data.id} notebook={query.data} />;
}
